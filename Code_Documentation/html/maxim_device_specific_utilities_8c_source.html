100 {
101  g_uchReadADCStop=0;
102  g_auSamples=auSamples;
103  g_unSampleSize=unSampleSize-1; // -1 because g_auCount counts from 0
104  g_auCount=0;
105 
106  g_auWriteBuffer[0] = (0x80 + (uchADCChannel<<4))<<8; //SPI read command for the selected ADC channel
107  g_auWriteBuffer[1] = 0;
108 
109  if(nSampleRate==0) //Timer setup for 1ksps
110  {
111  htim1.Init.Prescaler = 0; //Timer clock = 50MHz
112  htim1.Init.Period = 50000-1; //Period = 50MHz/50000
113  }
114  else if(nSampleRate==1) //Timer setup for 10ksps
115  {
116  htim1.Init.Prescaler = 0;
117  htim1.Init.Period = 5000-1; //Period = 50MHz/5000
118  }
119  else if(nSampleRate==2) //Timer setup for 25ksps
120  {
121  htim1.Init.Prescaler = 0;
122  htim1.Init.Period = 2000-1; //Period = 50MHz/2000
123  }
124  else if(nSampleRate==3) //Timer setup for 50ksps
125  {
126  htim1.Init.Prescaler = 0;
127  htim1.Init.Period = 1000-1; //Period = 50MHz/1000
128  }
129 
130  __HAL_SPI_RESET_CRC(&hspi1); //resets the SPI1 peripheral
131  __HAL_SPI_ENABLE(&hspi1); //enable the SPI1 peripheral
132 
133  g_uchReadADCStop=0; //block sampling is done when g_uchReadADCStop becomes 1
134  HAL_TIM_Base_Init(&htim1); //initializes timer with the above settings
135  HAL_TIM_Base_Start_IT(&htim1); //starts the timer and enables interrupt
136 
137  /*
138  * wait until the sampling is done
139  */
140  while(g_uchReadADCStop==0)
141  {
142  }
143  return unSampleSize;

147 void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
148 {
149  //Low-level commands are used to optimize the sampling rate
150 
151  HAL_GPIO_WritePin(GPIOA, GPIO_PIN_8, GPIO_PIN_RESET); //Set CS pin low
152  SPI1->DR = g_auWriteBuffer[0]; // write the first word of the SPI command to the SPI DR register
153  while(!__HAL_SPI_GET_FLAG(&hspi1,SPI_FLAG_TXE)); //wait until the TX empty flag asserts
154  SPI1->DR = g_auWriteBuffer[1]; // write the second word of the SPI command to the SPI DR register
155  while(!__HAL_SPI_GET_FLAG(&hspi1,SPI_FLAG_RXNE)); //wait until the RX not empty flag asserts
156  g_auSamples[g_auCount]=SPI1->DR; //read the DR register
157  while(!__HAL_SPI_GET_FLAG(&hspi1,SPI_FLAG_RXNE)); //wait until the RX not empty flag asserts
158 
159  HAL_GPIO_WritePin(GPIOA, GPIO_PIN_8, GPIO_PIN_SET); //Set CS pin high
160  g_auSamples[g_auCount]=SPI1->DR; //read the DR register for ADC data
161  if(g_auCount==g_unSampleSize)
162  {
163  g_uchReadADCStop=1;
164  HAL_TIM_Base_Stop_IT(&htim1); //stops the timer
165  }
166  /*if(__HAL_UART_GET_FLAG(&huart1,UART_FLAG_RXNE))
167  {
168  if(getchar()==0x1B)
169  {
170  g_uchReadADCStop=1;
171  HAL_TIM_Base_Stop_IT(&htim1); //stops the timer

174  g_auCount++;

220 void ADCInit(void)
231 {
232  // MAX1301 ADC Input Range Configuration
233  // Initialize the MAX1301 for AIN0=+/-10V, AIN1=+/-10V, AIN2=0 to 5V, AIN3=0 to 10V
234  //
235  // +/-10V input Votage Equation => -12.288 + (375uV x Code) = Input Voltage
236  // 4-20mA/0V to 5V input input Votage Equation => (93.75uV x Code) = Input Voltage
237  // 0 to 10V input input Votage Equation => (187.5uV x Code) = Input Voltage
238  //
239  // The higher four bits is the channel, 0x87 where 8 is AIN0
240  // The lower four bits is the input range, 0x87 where 7 is -10V to 10V
241  // Input Range Values are:
242  // 0xX3 => 0V to 5V
243  // 0xX4 => -5V to +5V
244  // 0xX6 => 0V to 10V
245  // 0xX7 => -10V to 10V
246  // 6VREF -10V to 10V AIN0 => 0x87
247  // 6VREF -10V to 10V AIN1 => 0x97
248  // 3VREF/2 0V to 5V AIN2 => 0xA3
249  // 3VREF 0V to 10V AIN3 => 0xB6
250
