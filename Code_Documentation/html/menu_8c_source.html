162 uint8_t menu_retrieve_keypress(uint32_t nUartAddress)
172 {
173  uint8_t uchInput;
174 
175  uchInput = getchar();
176 
177  if(uchInput==27) // Escape sequence (likely an arrow key)
178  {
179  uchInput = getchar();
180  if(uchInput==91) // Left bracket (part #2 of the 3 part escape sequence)
181  {
182  uchInput = getchar();
183  if(uchInput==65)
184  uchInput = KEYPRESS_ARROW_UP-10;
185  if(uchInput==66)
186  uchInput = KEYPRESS_ARROW_DOWN-10;
187  if(uchInput==67)
188  uchInput = KEYPRESS_ARROW_RIGHT-10;
189  if(uchInput==68)
190  uchInput = KEYPRESS_ARROW_LEFT-10;
191  if(uchInput==75)
192  uchInput = KEYPRESS_END - 10;
193  }
194  }
195  else if(uchInput>=48 && uchInput <=57) // digits 0..9
196  uchInput -=48; // subtract 48 to get the real number converted from the ASCII digit
197  else if(uchInput>=97 && uchInput<=122)
198  uchInput -=32; // convert lowercase to uppercase
199 
200  return(uchInput);
201 }
